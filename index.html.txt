    <script>
        // 支援觸控與滑鼠拖放
        function allowDrop(e) { e.preventDefault(); }
        function drag(e) { e.dataTransfer.setData("text", e.target.id); }
        function drop(e) {
            e.preventDefault();
            const id = e.dataTransfer.getData("text");
            const el = document.getElementById(id);
            if (e.target.classList.contains('pan') || e.target.classList.contains('inventory')) {
                e.target.appendChild(el);
                update();
            }
        }

        function update() {
            let L = 0, R = 0;
            document.querySelectorAll('#left-pan .weight').forEach(w => L += +w.dataset.v);
            document.querySelectorAll('#right-pan .weight').forEach(w => R += +w.dataset.v);
            
            // 限制最大傾斜度
            const angle = Math.max(-18, Math.min(18, (R - L) * 2));
            document.getElementById('moving-part').style.transform = `rotate(${angle}deg)`;
            
            const msg = document.getElementById('msg');
            if (L === R && L !== 0) msg.innerHTML = `🌟 平衡了！ (${L} = ${R})`;
            else if (L > R) msg.innerHTML = `⬅️ 左邊重 (${L} > ${R})`;
            else if (R > L) msg.innerHTML = `➡️ 右邊重 (${R} > ${L})`;
            else msg.innerHTML = "將砝碼拖到秤盤上";
        }
    </script>
